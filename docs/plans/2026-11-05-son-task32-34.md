**Step 1: Thêm tham số vào `rankAssetCandidates`**

Sửa signature và dùng `minTurnaround` thay vì hằng số:

```typescript
export function rankAssetCandidates(
  candidates: AssetCandidate[],
  startDay: Date,
  options: { minTurnaroundDays?: number } = {},
): RankedAssetCandidate[] {
  const minTurnaround = options.minTurnaroundDays ?? MIN_TURNAROUND_DAYS;
  return candidates
    .filter((candidate) => STATUS_RANK[candidate.status] !== undefined)
    .map((candidate) => {
      const usageCount = candidate.bookings.filter((b) => !NOT_COUNTED_FOR_WEAR.includes(b.status)).length;
      const tightTurnaround = candidate.bookings.some((b) => {
        if (!UNSETTLED_STATUSES.includes(b.status)) return false;
        if (b.rentalEndDate >= startDay) return false;
        const gapDays = Math.round((startDay.getTime() - b.rentalEndDate.getTime()) / DAY_MS);
        return gapDays < minTurnaround;  // ← dùng minTurnaround
      });
      return { ...candidate, usageCount, tightTurnaround };
    })
    .sort((a, b) =>
      Number(a.tightTurnaround) - Number(b.tightTurnaround) ||
      (STATUS_RANK[a.status] ?? 9) - (STATUS_RANK[b.status] ?? 9) ||
      a.usageCount - b.usageCount ||
      a.code.localeCompare(b.code),
    );
}
```

```typescript
import type { DamageFeeEntry } from "../admin/admin.constants";

/**
 * Tra cứu bảng định mức lỗi theo label và tính penalty.
 * @returns Số tiền phạt (VND), hoặc undefined nếu không tìm thấy label.
 */
export function resolvePenaltyFromTable(
  label: string,
  table: DamageFeeEntry[],
  garmentValue: number,
): number | undefined {
  const entry = table.find((e) => e.label === label);
  if (!entry) return undefined;
  if (entry.amount !== undefined) return entry.amount;
  if (entry.pct !== undefined) {
    return Math.round(((garmentValue * entry.pct) / 100) / 1000) * 1000;
  }
  return 0;
}

export function entryToSeverity(label: string, table: DamageFeeEntry[]): string {
  return table.find((e) => e.label === label)?.severity ?? "low";
}
```

**Step 5: Tích hợp vào `addFinding` trong `InspectionsService`**

Trong method `addFinding`, thêm logic tự điền:

```typescript
async addFinding(sessionId: string, dto: CreateFindingDto) {
  const finding = await this.prisma.runSerializable(async (tx) => {
    const session = await tx.inspectionSession.findUnique({
      where: { id: sessionId },
      include: {
        booking: { include: { items: true } },
        garmentAsset: { include: { garment_sizes: true } },
      },
    });
    if (!session) throw new NotFoundException("Inspection session not found.");
    if (!MUTABLE_INSPECTION_STATUSES.includes(session.status)) {
      throw new BadRequestException("Cannot add finding to a completed inspection.");
    }

    let resolvedPenalty = dto.penaltyAmount ?? 0;
    let resolvedSeverity = dto.severity ?? "low";

    // Tự điền penalty nếu staff không nhập và label khớp bảng định mức
    if (dto.penaltyAmount === undefined || dto.penaltyAmount === null) {
      const tableSetting = await tx.systemSetting.findUnique({ where: { key: "damage_fee_table" } });
      const feeTable: DamageFeeEntry[] = Array.isArray(tableSetting?.value)
        ? (tableSetting!.value as DamageFeeEntry[])
        : ((ADMIN_SETTING_DEFINITIONS.find((d) => d.key === "damage_fee_table")?.defaultValue as DamageFeeEntry[]) ?? []);

      if (feeTable.length > 0) {
        const matchedItem = session.booking?.items?.find((it) => it.garmentAssetId === session.garmentAssetId);
        const purchaseCost = Number(session.garmentAsset?.purchaseCost ?? 0);
        const depositAmount = Number(
          matchedItem?.depositAmount ?? (session.garmentAsset as any)?.garment_sizes?.deposit_amount ?? 0,
        );
        const dailyPrice = Number(matchedItem?.dailyPrice ?? (matchedItem as any)?.basePrice ?? 0);
        const garmentValue =
          purchaseCost > 0 ? purchaseCost
          : depositAmount > 0 ? depositAmount
          : dailyPrice > 0 ? dailyPrice * 4
          : 0;

        const autoAmount = resolvePenaltyFromTable(dto.findingType, feeTable, garmentValue);
        if (autoAmount !== undefined) {
          resolvedPenalty = autoAmount;
          resolvedSeverity = dto.severity ?? entryToSeverity(dto.findingType, feeTable);
        }
      }
    }

    return tx.inspectionFinding.create({
      data: {
        inspectionSessionId: sessionId,
        findingType: dto.findingType,
        severity: resolvedSeverity,
        description: dto.description ?? null,
        penaltyAmount: resolvedPenalty,
      },
    });
  });

  return ok({
    id: finding.id,
    findingType: finding.findingType,
```
